main:
  main:dec-def*

dec-def:
    func-def:type IDEN LPAREN params-def? RPAREN LBRACE line* RBRACE
    dec-assign:dec-assign
    type-def:type-def

type-def:
    struct-def:STRUCT IDEN LBRACE var-dec-colon* RBRACE

var-dec-colon:
    c:var-dec SEMI

var-dec-comma:
    dec-comma:COMMA var-dec-colon

line:
    control:control
    statement:statement-comma

control:
    if:IF LPAREN expr RPAREN block-or-line else?
    while:WHILE LPAREN expr RPAREN LBRACE line* RBRACE
    for:FOR LPAREN statement? SEMI statement? SEMI statement? RPAREN LBRACE block-or-line RBRACE

block-or-line:
    block:LBRACE line* RBRACE
    line:line

else:
    else:ELSE block-or-line

statement-comma:
    statement:statement SEMI

params-def:
    params:param-def comma-param-def*

comma-param-def:
    param:COMMA param-def

param-def:
    param:type variable-def

value:
    int:NUM
    str:STR
    char:CHAR
    cast:LPAREN type RPAREN value
    comp-lit:LPAREN type array-index-def* RPAREN LBRACE comp-entries? RBRACE

array-entries:
    entries:expr comma-expr*

comma-expr:
    comma-expr:COMMA expr

sub-variable-access:
    v:IDEN array-index-access*

inner-sub-variable-access:
    dot:FULL_STOP sub-variable-access
    arrow:RIGHT_ARROW sub-variable-access

variable-def:
    pointer:ASTERISKS variable-def
    variable:IDEN array-index-def*

array-index-access:
    array-index:LSQUARE NUM RSQUARE

array-index-def:
    array-index:LSQUARE NUM? RSQUARE

expr:
    sub-expr:sub-expr
    comp:comp-expr

sub-expr:
    value:value
    variable:sub-variable-access inner-sub-variable-access*
    parens:LPAREN expr RPAREN
    deref:ASTERISKS expr
    address-of:AMPERSAND expr
    func-call:IDEN LPAREN params? RPAREN

comp-expr:
    comp-expr:sub-expr operator expr

operator:
    less:LESS_THAN
    equal:EEQUALS
    plus:PLUS
    minus:MINUS
    assign:EQUALS

statement:
    dec-assign:dec-assign
    var-dec:var-dec
    expr:expr
    return:RETURN expr

var-dec:
    var-dec:type variable-def

dec-assign:
    standard:type variable-def EQUALS initialiser

initialiser:
    expr:expr
    list:LBRACE comp-entries? RBRACE

comp-entries:
    entries:comp-entry comma-comp-entry* COMMA?

comma-comp-entry:
    e:COMMA comp-entry

comp-entry:
    labelled:entry-label-field* EQUALS initialiser
    anon:initialiser

entry-label-field:
    c:FULL_STOP IDEN

params:
    params:expr comma-expr*

comma-expr:
    comma-expr:COMMA expr

type:
    int:INT_TYPE
    unsigned-int:UNSIGNED INT_TYPE
    char:CHAR_TYPE
    void:VOID
    custom:IDEN
    struct:STRUCT IDEN
